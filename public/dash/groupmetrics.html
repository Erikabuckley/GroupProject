import matplotlib.pyplot as plt import numpy as np from js import fetch from pyscript import display import js import json import asyncio from datetime import datetime # retrieve points data async def get_points_data(): res = await fetch("/updatePointsDateGroup") data = (await res.json()).to_py() dates = [] points = [] for row in data: dates.append(datetime.strptime(row["date"], "%Y-%m-%d")) points.append(row["total"]) return dates, points # make points bar chart async def make_points_chart(): dates, points = await get_points_data() fig, ax = plt.subplots(figsize=(4,3)) ax.bar(dates, points, color=(0.6, 0.8, 0.7)) ax.set_ylim(bottom=0) ax.set_xlabel("Date") ax.set_ylabel("Number of points earned") ax.set_title("Points Earned Over Time") plt.xticks(rotation=45) plt.tight_layout() display(fig, target="graph1", append=False) async def get_co2_data(): res = await fetch("/updateActionDatesGroup") data = (await res.json()).to_py() dates = [] carbon = [] for row in data: dates.append(datetime.strptime(row["date"], "%Y-%m-%d")) carbon.append(row["calculated_co2e"]) return dates, carbon async def make_co2_scatter(): dates, carbon = await get_co2_data() fig, ax = plt.subplots(figsize=(4,3)) ax.scatter(dates, carbon, color=(0.6, 0.8, 0.7)) ax.set_xlabel("Date") ax.set_ylabel("CO2 Saved (g)") ax.set_title("Carbon Emissions Saved By Group") plt.xticks(rotation=45) plt.tight_layout() display(fig, target="graph2", append=False) # retrieve action type data async def get_action_type_data(): res = await fetch("/updateActionTypesGroup") data = (await res.json()).to_py() emissions = data["carbon"] action_types = data["category"] return emissions, action_types async def make_action_type_pie(): emissions, action_types = await get_action_type_data() my_colours = [(0.6, 0.8, 0.7), (0.4, 0.6, 0.7), (0.6, 0.7, 0.6)] fig, ax = plt.subplots(figsize=(4,3)) ax.pie(emissions, labels = action_types, colors = my_colours) ax.set_title("Carbon Emissions Saved Per Action Type") display(fig, target="graph3", append=False) asyncio.ensure_future(make_co2_scatter()) asyncio.ensure_future(make_points_chart()) asyncio.ensure_future(make_action_type_pie())
